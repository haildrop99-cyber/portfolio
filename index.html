Code

외부 JS 라이브러리 없이 브라우저 기본 API만으로 인터랙션을 만들었습니다.

// 숫자 카운트업 — 화면에 60% 이상 보일 때 한 번만 실행
const cio = new IntersectionObserver(es => es.forEach(e => {
  if (!e.isIntersecting) return;
  cio.unobserve(e.target);
  const el = e.target, to = +el.dataset.to, t0 = performance.now();
  (function tick(t){
    const p = Math.min((t - t0) / 1400, 1);
    el.textContent = Math.round(to * (1 - Math.pow(1 - p, 3)));   // ease-out cubic
    if (p < 1) requestAnimationFrame(tick);
  })(t0);
}), { threshold: .6 });

      Key points

      코드는 저장소의 실제 코드에서 발췌했습니다.

      // _worker.js — 매장 코드는 D1에만 있고 공개 API 응답에는 포함하지 않는다
if (p === '/api/redeem' && request.method === 'POST') {
  const { phone, storeKey, code } = await request.json();
  if (!storeKey || !code) return json({ error: 'storeKey/code required' }, 400);

  const row = await env.DB
    .prepare(`SELECT name, code FROM stores WHERE key = ?1`)
    .bind(storeKey).first();
  if (!row) return json({ error: 'store not found' }, 404);

  if (String(code).trim().toUpperCase() !== String(row.code).trim().toUpperCase())
    return json({ ok: false, reason: 'invalid_code' });

  await logEvent(env, phone, 'stamp', row.name, '코드인증');  // 서버가 스탬프를 기록
  return json({ ok: true });
}
